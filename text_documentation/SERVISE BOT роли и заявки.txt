# роли:
    #пользователь(планшет на филиале/акк менеджера смены)
        # оповещение о проблеме идет сначала упр и ожидает подтверждения (кроме критических ситуаций)
        # нет возможности передачи ролей
        # может только создать заявку
        # видеть статус своих заявок
        # НЕ менять статус
        # НЕ редактировать после отправки
    #управляющий филиалом
        # полный функционал бота
        # есть возможность добавить роль пользователя
        # может подтвердить заявку от пользователя и передать дальше
        # может создать и подтвердить выполнение заявки (создать, закрыть, невыполнена)
        # просмотр заявок филиала
        #если заявка частична, то есть выбор закрыть и пересоздать или поставить статус невыполнена
    #отдел
        #предложение добавления коллег (без финального права) -> ожидание подтв руководителя
        #смена статусов (в работе / ожидание / выполнено / отклонено)
        #передать заявку руководителю если невозможно выполнить. (холодильник сломался, починить нельзя, нужен новый)
    #руководитель
        # доступ к статистике
        # возможность выдачи роли пользователь, отдел, упр
        # возможность просмотра всех текущих заявок
        # возможность создания заявки с нужным уровнем срочности
        # принудительное закрытие заявок
    #аналитик
        # доступ к статистике
        # отчёты
        # экспорт данных
        # без доступа к заявкам

#Статусы заявки:
    # создана без подтверждения (для пользователя)
    # создана (выдает упр и руководитель, переход заявки пользователя после подтверждения упр)
    # в работе(переключает отдел) означает что заявка принята
    # ожидание(переключает отдел) означает что для выполнения нужно время. допустим коммент
    # выполнено(переключает отдел) означает что работа выполнена
    # ожидает подтверждения(переключает бот) ждем подтверждения управляющего
    # отклонено(переключает отдел) означает что заявку невозможно выполнить. (тут нужна метка в статистике)
    # закрыта(переключает упр) означает что заявка выполнена
